local function create(class, properties, parent)
    local object = Instance.new(class)
    for key, value in pairs(properties or {}) do
        object[key] = value
    end
    object.Parent = parent
    return object
end

local function corner(parent, radius)
    return create("UICorner", {CornerRadius = UDim.new(0, radius)}, parent)
end

local function animate(window, object, properties, duration, style)
    if window.dead or not object then
        return
    end
    local active = window.animations[object]
    if active then
        local cancelled = {}
        for property in pairs(properties) do
            if active[property] then
                cancelled[active[property]] = true
            end
        end
        for tween in pairs(cancelled) do
            tween:Cancel()
            for property, current in pairs(active) do
                if current == tween then
                    active[property] = nil
                end
            end
        end
    end
    if window.motion == false or duration == 0 then
        for property, value in pairs(properties) do
            object[property] = value
        end
        return
    end
    active = active or {}
    window.animations[object] = active
    local tween = tweens:Create(object, TweenInfo.new(duration or 0.2, style or Enum.EasingStyle.Quart, Enum.EasingDirection.Out), properties)
    for property in pairs(properties) do
        active[property] = tween
    end
    tween.Completed:Connect(function()
        for property, current in pairs(active) do
            if current == tween then
                active[property] = nil
            end
        end
        if next(active) == nil and window.animations[object] == active then
            window.animations[object] = nil
        end
    end)
    tween:Play()
    return tween
end

local function paint(window, object, property, color, duration)
    if not window.paints[object] then
        window.paints[object] = {}
        object.Destroying:Connect(function()
            window.paints[object] = nil
        end)
    end
    window.paints[object][property] = color
    local value = window.colors[color]
    if value then
        animate(window, object, {[property] = value}, duration or 0)
    end
end

end

local function glyph(value)
    if type(value) == "number" then
        return "rbxassetid://" .. tostring(value)
    end
    if type(value) ~= "string" or value == "" then
        return ""
    end
    if value:match("^%d+$") then
        return "rbxassetid://" .. value
    end
    if value:match("^rbxasset") or value:match("^https?://") then
        return value
    end
    local name = value:lower():gsub("^lucide:", "")
    return icons[name] and ("rbxassetid://" .. tostring(icons[name])) or ""
end

local function picture(window, parent, value, color)
    local object = create("ImageLabel", {
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        Image = glyph(value),
        ScaleType = Enum.ScaleType.Fit
    }, parent)
    paint(window, object, "ImageColor3", color or "text")
    return object
end

local function hover(window, object, normal, over)
    object.MouseEnter:Connect(function()
        paint(window, object, "BackgroundColor3", over, 0.14)
    end)
    object.MouseLeave:Connect(function()
        paint(window, object, "BackgroundColor3", normal, 0.14)
    end)
end

local function copy(value)
    if typeof(value) ~= "table" then
        return value
    end
    local result = {}
    for key, item in pairs(value) do
        result[key] = copy(item)
    end
    return result
end

local function equal(first, second)
    if typeof(first) ~= "table" or typeof(second) ~= "table" then
        return first == second
    end
    for key, value in pairs(first) do
        if not equal(value, second[key]) then
            return false
        end
    end
    for key in pairs(second) do
        if first[key] == nil then
            return false
        end
    end
    return true
end

local function fire(callback, ...)
    if type(callback) ~= "function" then
        return
    end
    local arguments = table.pack(...)
    task.defer(function()
        local ok, message = pcall(callback, table.unpack(arguments, 1, arguments.n))
        if not ok then
            warn("[Glacier] callback: " .. tostring(message))
        end
    end)
end

local function schedule(window, delay, callback)
    local token = {}
    window.jobs[token] = true
    token.thread = task.delay(delay, function()
        if not window.jobs[token] then
            return
        end
        window.jobs[token] = nil
        if not window.dead then
            callback()
        end
    end)
    return token
end

local function cancel(window, token)
    if not token or not window.jobs[token] then
        return
    end
    window.jobs[token] = nil
    if token.thread and type(task.cancel) == "function" then
        pcall(task.cancel, token.thread)
    end
end

local function commit(window, handle, value, callback, silent)
    local changed = not equal(handle.value, value)
    handle.value = copy(value)
    if handle.flag then
        window.flags[handle.flag] = copy(value)
    end
    if changed and not silent then
        fire(callback, copy(value))
    end
end

local function verify(window, options)
    local flag = options.flag
    if flag ~= nil then
        assert(type(flag) == "string" and flag ~= "", "Glacier: flag must be a non-empty string")
        assert(window.controls[flag] == nil, "Glacier: duplicate flag '" .. flag .. "'")
    end
end

local function register(window, options, handle, value)
    verify(window, options)
    local flag = options.flag
    if flag ~= nil then
        handle.flag = flag
        window.controls[flag] = handle
    end
    handle:set(value, true)
    handle.default = copy(handle.value)
end

local function keycode(value)
    if typeof(value) == "EnumItem" and value.EnumType == Enum.KeyCode then
        return value.Name ~= "Unknown" and value.Name ~= "None" and value or nil
    end
    if type(value) == "string" and value ~= "" then
        local ok, result = pcall(function()
            return Enum.KeyCode[value]
        end)
        if ok and result and result.Name ~= "Unknown" and result.Name ~= "None" then
            return result
        end
    end
    return nil
end

local function position(value)
    if type(value) ~= "string" then
        return nil
    end
    local name = value:lower():gsub("[%s_]", "-"):gsub("%-+", "-")
    local aliases = {
        left = "center-left",
        right = "center-right",
        top = "top-center",
        bottom = "bottom-center",
        ["left-top"] = "top-left",
        ["right-top"] = "top-right",
        ["left-bottom"] = "bottom-left",
        ["right-bottom"] = "bottom-right"
    }
    name = aliases[name] or name
    return positions[name] and name or nil
end

local function capture(window, object, change, finish)
    if object.UserInputType ~= Enum.UserInputType.MouseButton1 and object.UserInputType ~= Enum.UserInputType.Touch then
        return
    end
    if window.pointer and window.pointer.finish then
        window.pointer.finish()
    end
    window.pointer = {object = object, change = change, finish = finish}
    change(object.Position)
end

local function flow(tab)
    local narrow = tab.scroll.AbsoluteSize.X < 620
    if tab.narrow ~= narrow then
        tab.narrow = narrow
        for _, section in ipairs(tab.sections) do
            section.frame.Parent = narrow and tab.left or section.side == "right" and tab.right or tab.left
        end
    end
    local left = tab.leftlist.AbsoluteContentSize.Y
    local right = narrow and 0 or tab.rightlist.AbsoluteContentSize.Y
    tab.left.Size = UDim2.new(narrow and 1 or 0.5, narrow and 0 or -9, 0, left)
    tab.right.Visible = not narrow
    tab.right.Position = UDim2.new(0.5, 9, 0, 0)
    tab.right.Size = UDim2.new(0.5, -9, 0, right)
    tab.scroll.CanvasSize = UDim2.new(0, 0, 0, math.max(left, right) + 12)
end

local function filter(window)
    local query = window.query:lower()
    for _, tab in ipairs(window.tabs) do
        local found = false
        for _, section in ipairs(tab.sections) do
            local matches = query == "" or section.name:lower():find(query, 1, true) ~= nil
            local visible = matches
            for _, entry in ipairs(section.entries) do
                local show = matches or entry.text:lower():find(query, 1, true) ~= nil
                entry.frame.Visible = show
                visible = visible or show
            end
            section.frame.Visible = visible
            local forced = query ~= "" and visible
            if section.forced ~= forced then
                section.forced = forced
                local expanded = section.expanded or forced
                section.rule.Visible = expanded
                animate(window, section.arrow, {Rotation = expanded and 90 or 0}, 0.18)
                section.resize()
            end
            found = found or visible
        end
        tab.empty.Visible = query ~= "" and not found
    end
end

local function entry(section, frame, value)
    local window = section.window
    section.sequence += 1
    frame.LayoutOrder = section.sequence
    local record = {frame = frame, text = tostring(value or "")}
    table.insert(section.entries, record)
    local handle = {record = record, section = section}
    function handle:remove()
        if self.dead then
            return
        end
        self.dead = true
        if self.flag and window.controls[self.flag] == self then
            window.controls[self.flag] = nil
            window.flags[self.flag] = nil
        end
        if window.listening == self then
            window.listening = nil
        end
        if self.held then
            self.held = false
            fire(self.callback, false, self.value)
        end
        window.pointer = nil
        for index, item in ipairs(window.binds) do
            if item == self then
                table.remove(window.binds, index)
                break
            end
        end
        for index, item in ipairs(section.entries) do
            if item == record then
                table.remove(section.entries, index)
                break
            end
        end
        frame:Destroy()
        filter(window)
    end
    filter(window)
    return handle
end

local function row(section, height, name)
    local frame = create("Frame", {

        BorderSizePixel = 0,
        ClipsDescendants = true,
        Size = UDim2.new(1, 0, 0, height)
    }, section.body)
    return frame, entry(section, frame, name)
end

local function pack(value)
    local kind = typeof(value)
    if value == nil then
        return {kind = "none"}
    elseif kind == "Color3" then
        return {kind = "color", value = value:ToHex()}
    elseif kind == "EnumItem" then
        return {kind = "key", value = value.Name}
    elseif type(value) == "table" then
        local result = {}
        for key, item in pairs(value) do
            result[key] = pack(item)
        end
        return result
    end
    return value
end

local function unpack(value)
    if type(value) ~= "table" then
        return value
    end
    if value.kind == "none" then
        return nil
    elseif value.kind == "color" then
        if type(value.value) ~= "string" or not value.value:match("^%x%x%x%x%x%x$") then
            error("invalid color")
        end
        return Color3.fromHex(value.value)
    elseif value.kind == "key" then
        local result = keycode(value.value)
        if not result then
            error("invalid key")
        end
        return result
    end
    local result = {}
    for key, item in pairs(value) do
        result[key] = unpack(item)
    end
    return result
end

local function safe(value)
    return type(value) == "string" and #value > 0 and #value <= 64 and value:match("^[%w_%-]+$") ~= nil
end

function windowmethods:theme(value)
    if value == nil then
        return copy(self.colors)
    end
    if type(value) == "string" then
        local accent = presets[value:lower()]
        if accent then
            value = {accent = accent}
        else
            return false, "unknown theme preset"
        end
    end
    if type(value) ~= "table" then
        return false, "theme must be a table or preset name"
    end
    for name, color in pairs(value) do
        if palette[name] and typeof(color) == "Color3" then
            self.colors[name] = color
        end
    end
    for object, properties in pairs(self.paints) do
        if object.Parent then
            for property, name in pairs(properties) do
                if self.colors[name] then
                    animate(self, object, {[property] = self.colors[name]}, 0)
                end
            end
        end
    end
    return true
end

function windowmethods:configure(options)
    if type(options) ~= "table" then
        return self
    end
    if options.theme ~= nil then
        self:theme(options.theme)
    end
    if options.motion ~= nil then
        self.motion = options.motion ~= false
    end
    if options.key ~= nil then
        self.key = keycode(options.key)
    end
    if options.reopen ~= nil then
        self.reopen = options.reopen ~= false
        self.floating.Visible = not self.frame.Visible and self.reopen
    end
    if safe(options.folder) then
        self.folder = options.folder
    end
    if type(options.notifications) == "table" then
        local settings = options.notifications
        local target = position(settings.position)
        if target then
            self.notifications.position = target
        end
        if type(settings.duration) == "number" and settings.duration == settings.duration then
            self.notifications.duration = math.clamp(settings.duration, 0, 3600)
        end
        if type(settings.limit) == "number" and settings.limit == settings.limit then
            self.notifications.limit = math.clamp(math.floor(settings.limit), 1, 10)
        end
    end
    return self
end

function windowmethods:visible(value)
    if value == nil then
        return self.frame.Visible
    end
    if self.dead then
        return self
    end
    local show = value == true
    self.frame.Visible = show
    self.floating.Visible = not show and self.reopen
    if show then
        self.scale.Scale = self.motion and 0.975 or 1
        animate(self, self.scale, {Scale = 1}, 0.2)
    else
        local focused = input:GetFocusedTextBox()
        if focused and focused:IsDescendantOf(self.frame) then
            focused:ReleaseFocus()
        end
        if self.listening and self.listening.cancel then
            self.listening:cancel()
        end
        self.pointer = nil
    end
    return self
end

function windowmethods:open()
    return self:visible(true)
end

function windowmethods:close()
    return self:visible(false)
end

function windowmethods:search(value)
    self.searchbox.Text = tostring(value or "")
    self.query = self.searchbox.Text
    filter(self)
    return self
end

function windowmethods:destroy()
    if self.dead then
        return
    end
    self.dead = true
    local focused = input:GetFocusedTextBox()
    if focused and focused:IsDescendantOf(self.frame) then
        focused:ReleaseFocus()
    end
    for _, binding in ipairs(self.binds) do
        if binding.held then
            binding.held = false
            fire(binding.callback, false, binding.value)
        end
    end
    self.pointer = nil
    self.listening = nil
    for _, list in pairs(self.stacks) do
        while #list.items > 0 do
            list.items[1]:close()
        end
    end
    for token in pairs(self.jobs) do
        cancel(self, token)
    end
    local active = {}
    for _, properties in pairs(self.animations) do
        for _, tween in pairs(properties) do
            active[tween] = true
        end
    end
    for tween in pairs(active) do
        tween:Cancel()
    end
    for _, connection in ipairs(self.connections) do
        connection:Disconnect()
    end
    self.gui:Destroy()
    table.clear(self.animations)
    table.clear(self.paints)
    for index, item in ipairs(windows) do
        if item == self then
            table.remove(windows, index)
            break
        end
    end
end

function windowmethods:tab(options)
    options = type(options) == "table" and options or {name = options}
    local window = self
    local name = tostring(options.name or ("Tab " .. tostring(#window.tabs + 1)))
    local navigation = button(window.navigation)
    navigation.Name = name
    navigation.Size = UDim2.new(1, 0, 0, 44)
    navigation.LayoutOrder = #window.tabs + 1
    corner(navigation, 12)
    paint(window, navigation, "BackgroundColor3", "accent")
    local symbol = picture(window, navigation, options.icon or "grid", "muted")
    symbol.AnchorPoint = Vector2.new(0.5, 0.5)
    symbol.Position = UDim2.new(0.5, 0, 0.5, 0)
    symbol.Size = UDim2.fromOffset(21, 21)
    local group = create("CanvasGroup", {
        Name = name,
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        Size = UDim2.fromScale(1, 1),
        Visible = false
    }, window.workspace)
    local scroll = create("ScrollingFrame", {
        Name = "Content",
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        Size = UDim2.fromScale(1, 1),
        CanvasSize = UDim2.new(),
        ScrollBarThickness = 3,
        ScrollingDirection = Enum.ScrollingDirection.Y,
        ElasticBehavior = Enum.ElasticBehavior.WhenScrollable
    }, group)
    paint(window, scroll, "ScrollBarImageColor3", "border")
    local left = create("Frame", {Name = "Left", BackgroundTransparency = 1, BorderSizePixel = 0}, scroll)
    local right = create("Frame", {Name = "Right", BackgroundTransparency = 1, BorderSizePixel = 0}, scroll)
    local leftlist = create("UIListLayout", {SortOrder = Enum.SortOrder.LayoutOrder, Padding = UDim.new(0, 16)}, left)
    local rightlist = create("UIListLayout", {SortOrder = Enum.SortOrder.LayoutOrder, Padding = UDim.new(0, 16)}, right)
    local empty = label(window, group, "No matching controls", 14, "medium", "muted")
    empty.Size = UDim2.new(1, 0, 0, 56)
    empty.TextXAlignment = Enum.TextXAlignment.Center
    empty.Visible = false
    local tab = setmetatable({
        window = window,
        name = name,
        navigation = navigation,
        symbol = symbol,
        group = group,
        scroll = scroll,
        left = left,
        right = right,
        leftlist = leftlist,
        rightlist = rightlist,
        empty = empty,
        sections = {},
        sequence = 0
    }, {__index = tabmethods})
    table.insert(window.tabs, tab)
    leftlist:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
        flow(tab)
    end)
    rightlist:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
        flow(tab)
    end)
    scroll:GetPropertyChangedSignal("AbsoluteSize"):Connect(function()
        flow(tab)
    end)
    navigation.Activated:Connect(function()
        tab:select()
    end)
    navigation.MouseEnter:Connect(function()
        if window.active ~= tab then
            paint(window, navigation, "BackgroundColor3", "raised", 0.14)
            animate(window, navigation, {BackgroundTransparency = 0.25}, 0.14)
        end
        if input.MouseEnabled then
            local x = navigation.AbsolutePosition.X - window.frame.AbsolutePosition.X + navigation.AbsoluteSize.X + 10
            local y = math.clamp(navigation.AbsolutePosition.Y - window.frame.AbsolutePosition.Y + 7, 8, math.max(8, window.frame.AbsoluteSize.Y - 38))
            window.tooltip.Text = name
            window.tooltip.Position = UDim2.fromOffset(x, y)
            window.tooltip.Visible = true
        end
    end)
    navigation.MouseLeave:Connect(function()
        window.tooltip.Visible = false
        if window.active ~= tab then
            animate(window, navigation, {BackgroundTransparency = 1}, 0.14)
        end
    end)
    flow(tab)
    if not window.active then
        tab:select()
    else
        navigation.BackgroundTransparency = 1
    end
    return tab
end

function tabmethods:select()
    local window = self.window
    if window.dead then
        return self
    end
    window.active = self
    window.heading.Text = self.name
    for _, tab in ipairs(window.tabs) do
        local selected = tab == self
        tab.group.Visible = selected
        paint(window, tab.navigation, "BackgroundColor3", selected and "accent" or "raised", 0.16)
        animate(window, tab.navigation, {BackgroundTransparency = selected and 0.04 or 1}, 0.16)
        paint(window, tab.symbol, "ImageColor3", selected and "accenttext" or "muted", 0.16)
        if selected then
            tab.group.GroupTransparency = window.motion and 1 or 0
            animate(window, tab.group, {GroupTransparency = 0}, 0.2)
            flow(tab)
        end
    end
    filter(window)
    return self
end

function tabmethods:section(options)
    options = type(options) == "table" and options or {name = options}
    local tab = self
    local window = tab.window
    tab.sequence += 1
    local side = tostring(options.side or (tab.sequence % 2 == 1 and "left" or "right")):lower()
    if side == "2" then
        side = "right"
    end
    if side ~= "right" then
        side = "left"
    end
    local frame = create("Frame", {
        Name = tostring(options.name or "Section"),
        BackgroundTransparency = 0,
        BorderSizePixel = 0,
        ClipsDescendants = true,
        LayoutOrder = tab.sequence,
        Size = UDim2.new(1, 0, 0, 60)
    }, tab.narrow and tab.left or side == "right" and tab.right or tab.left)
    corner(frame, 15)
    paint(window, frame, "BackgroundColor3", "surface")
    outline(window, frame, 0.23)
    local header = button(frame)
    header.Name = "Header"
    header.Size = UDim2.new(1, 0, 0, 60)
    local tile = create("Frame", {
        BackgroundTransparency = 0,
        BorderSizePixel = 0,
        Position = UDim2.fromOffset(16, 13),
        Size = UDim2.fromOffset(34, 34)
    }, header)
    corner(tile, 10)
    paint(window, tile, "BackgroundColor3", "raised")
    local symbol = picture(window, tile, options.icon or "layers", "accent")
    symbol.AnchorPoint = Vector2.new(0.5, 0.5)
    symbol.Position = UDim2.fromScale(0.5, 0.5)
    symbol.Size = UDim2.fromOffset(19, 19)
    local heading = label(window, header, options.name or "Section", 15, "bold", "text")
    heading.Position = UDim2.fromOffset(62, 0)
    heading.Size = UDim2.new(1, -100, 1, 0)
    local arrow = picture(window, header, "chevron", "muted")
    arrow.AnchorPoint = Vector2.new(1, 0.5)
    arrow.Position = UDim2.new(1, -17, 0.5, 0)
    arrow.Size = UDim2.fromOffset(15, 15)
    local rule = create("Frame", {
        BackgroundTransparency = 0.55,
        BorderSizePixel = 0,
        Position = UDim2.new(0, 16, 0, 59),
        Size = UDim2.new(1, -32, 0, 1)
    }, frame)
    paint(window, rule, "BackgroundColor3", "border")
    local body = create("Frame", {
        Name = "Elements",
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        Position = UDim2.fromOffset(16, 68),
        Size = UDim2.new(1, -32, 0, 0)
    }, frame)
    local list = create("UIListLayout", {SortOrder = Enum.SortOrder.LayoutOrder, Padding = UDim.new(0, 4)}, body)
    local section = setmetatable({
        window = window,
        tab = tab,
        frame = frame,
        body = body,
        list = list,
        arrow = arrow,
        rule = rule,
        heading = heading,
        name = tostring(options.name or "Section"),
        side = side,
        entries = {},
        sequence = 0,
        expanded = options.open ~= false,
        forced = false,
        collapsible = options.collapsible ~= false
    }, {__index = sectionmethods})
    table.insert(tab.sections, section)
    local function resize()
        local height = list.AbsoluteContentSize.Y
        body.Size = UDim2.new(1, -32, 0, height)
        local target = (section.expanded or section.forced) and 68 + height + 16 or 60
        animate(window, frame, {Size = UDim2.new(1, 0, 0, target)}, 0.18)
    end
    section.resize = resize
    list:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(resize)
    header.Activated:Connect(function()
        if section.collapsible then
            section:open(not section.expanded)
        end
    end)
    arrow.Visible = section.collapsible
    arrow.Rotation = section.expanded and 90 or 0
    rule.Visible = section.expanded
    resize()
    flow(tab)
    filter(window)
    return section
end

function sectionmethods:open(value)
    self.expanded = value ~= false
    local expanded = self.expanded or self.forced
    self.rule.Visible = expanded
    animate(self.window, self.arrow, {Rotation = expanded and 90 or 0}, 0.18)
    self.resize()
    return self
end

function sectionmethods:label(options)
    options = type(options) == "table" and options or {text = options}
    local window = self.window
    local text = tostring(options.text or "Label")
    local frame, handle = row(self, 36, text)
    local object = label(window, frame, text, options.size or 13, options.bold and "bold" or nil, options.color or "muted")
    object.Size = UDim2.fromScale(1, 1)
    object.TextWrapped = options.wrap == true
    if options.wrap then
        object.TextTruncate = Enum.TextTruncate.None
        object.AutomaticSize = Enum.AutomaticSize.Y
        object.Size = UDim2.new(1, 0, 0, 0)
        frame.AutomaticSize = Enum.AutomaticSize.Y
        frame.Size = UDim2.new(1, 0, 0, 0)
    end
    function handle:set(value)
        object.Text = tostring(value or "")
        self.record.text = object.Text
        filter(window)
        return self
    end
    function handle:update(value)
        return self:set(value)
    end
    function handle:get()
        return object.Text
    end
    return handle
end

    body.TextTruncate = Enum.TextTruncate.None
    body.TextYAlignment = Enum.TextYAlignment.Top
    function handle:update(value)
        if type(value) == "table" then
            if value.title ~= nil then
                heading.Text = tostring(value.title)

            body.Text = tostring(value or "")
        end

function sectionmethods:slider(options)
    options = type(options) == "table" and options or {name = options}
    local window = self.window
    verify(window, options)
    local name = tostring(options.name or "Slider")
    local minimum = tonumber(options.min) or 0
    local maximum = tonumber(options.max) or 100
    local step = tonumber(options.step) or 1
    assert(minimum == minimum and maximum == maximum and minimum > -math.huge and maximum < math.huge and maximum > minimum, "Glacier: slider max must exceed min")
    if step ~= step or step <= 0 or step == math.huge then
        step = 1
    end
    step = math.max(step, 0.000001)
    local digits = tonumber(options.decimals)
    if digits then
        digits = math.clamp(math.floor(digits), 0, 6)
    else
        local fraction = string.format("%.6f", step):match("%.(%d+)"):gsub("0+$", "")
        digits = #fraction
    end
    local frame, handle = row(self, 76, name)
    handle.kind = "number"
    title(window, frame, name, 1).Size = UDim2.new(1, -80, 0, 38)
    local amount = label(window, frame, "", 12, "medium", "muted")
    amount.AnchorPoint = Vector2.new(1, 0)
    amount.Position = UDim2.new(1, 0, 0, 0)
    amount.Size = UDim2.fromOffset(76, 38)
    amount.TextXAlignment = Enum.TextXAlignment.Right
    local hit = button(frame)
    hit.Position = UDim2.fromOffset(0, 34)
    hit.Size = UDim2.new(1, 0, 0, 40)
    local track = create("Frame", {
        AnchorPoint = Vector2.new(0, 0.5),
        BorderSizePixel = 0,
        Position = UDim2.new(0, 9, 0.5, 0),
        Size = UDim2.new(1, -18, 0, 6)
    }, hit)
    corner(track, 4)
    paint(window, track, "BackgroundColor3", "control")
    local fill = create("Frame", {BorderSizePixel = 0, Size = UDim2.fromScale(0, 1)}, track)
    corner(fill, 4)
    paint(window, fill, "BackgroundColor3", "accent")
    local knob = create("Frame", {
        AnchorPoint = Vector2.new(0.5, 0.5),
        BackgroundColor3 = Color3.fromRGB(243, 247, 249),
        BorderSizePixel = 0,
        Position = UDim2.new(0, 0, 0.5, 0),
        Size = UDim2.fromOffset(17, 17)
    }, track)
    corner(knob, 9)
    outline(window, knob, 0.1)
    function handle:set(value, silent)
        value = tonumber(value)
        if not value or value ~= value or value == math.huge or value == -math.huge then
            return self
        end
        value = math.clamp(value, minimum, maximum)
        if value < maximum then
            value = math.clamp(minimum + math.floor((value - minimum) / step + 0.5) * step, minimum, maximum)
        end
        value = tonumber(string.format("%.6f", value))
        local fraction = (value - minimum) / (maximum - minimum)
        amount.Text = tostring(options.prefix or "") .. string.format("%." .. tostring(digits) .. "f", value) .. tostring(options.suffix or "")
        commit(window, self, value, options.callback, silent)
        animate(window, fill, {Size = UDim2.fromScale(fraction, 1)}, 0.1)
        animate(window, knob, {Position = UDim2.new(fraction, 0, 0.5, 0)}, 0.1)
        return self
    end
    function handle:get()
        return self.value
    end
    hit.InputBegan:Connect(function(object)
        capture(window, object, function(point)
            local fraction = math.clamp((point.X - track.AbsolutePosition.X) / math.max(1, track.AbsoluteSize.X), 0, 1)
            handle:set(minimum + (maximum - minimum) * fraction)
        end)
    end)
    local initial = tonumber(options.default)
    if not initial or initial ~= initial or initial == math.huge or initial == -math.huge then
        initial = minimum
    end
    register(window, options, handle, initial)
    return handle
end

            for _, item in ipairs(items) do
                if type(item) == "string" or type(item) == "number" then
                    local text = tostring(item)
                    if not seen[text] then
                        seen[text] = true
                        table.insert(result, text)
                    end
                end
            end
        end
        return result
    end
    local function contains(value)
        for _, item in ipairs(values) do
            if item == value then

            end
        end

    end

        if multiple then

            if type(value) == "table" then
                for _, item in ipairs(value) do
                    desired[tostring(item)] = true
                end
            end

            for _, item in ipairs(values) do
                if desired[item] then
                    table.insert(result, item)
                end
            end
            value = result
        else
            value = value ~= nil and tostring(value) or nil
            if value and not contains(value) then
                value = nil
            end
        end
        commit(window, self, value, options.callback, silent)
        refresh()
        return self
    end
    function handle:get()
        return copy(self.value)
    end
    function handle:open(value)
        opened = value ~= false
        resize()
        return self
    end
    function handle:update(items, silent)
        values = clean(items)
        self:set(self.value, silent)
        rebuild()
        return self
    end
    control.Activated:Connect(function()
        handle:open(not opened)
    end)
    values = clean(options.options)
    register(window, options, handle, multiple and (type(options.default) == "table" and options.default or {}) or options.default)
    rebuild()
    return handle
end

local function valid(handle, value)
    if handle.kind == "boolean" then
        return type(value) == "boolean"
    elseif handle.kind == "number" then
        return type(value) == "number" and value == value and value ~= math.huge and value ~= -math.huge
    elseif handle.kind == "string" then
        return type(value) == "string"
    elseif handle.kind == "color" then
        return typeof(value) == "Color3"
    elseif handle.kind == "key" then
        return value == nil or keycode(value) ~= nil
    elseif handle.kind == "dropdown" then
        if not handle.multi then
            return value == nil or type(value) == "string"
        end
        if type(value) ~= "table" then
            return false
        end
        for key, item in pairs(value) do
            if type(key) ~= "number" or type(item) ~= "string" then
                return false
            end
        end
        return true
    end
    return false
end

function windowmethods:export()
    local data = {
        version = 1,
        flags = {},
        theme = {},
        notifications = copy(self.notifications),
        key = pack(self.key)
    }
    for flag, handle in pairs(self.controls) do
        data.flags[flag] = pack(handle.value)
    end
    for name, color in pairs(self.colors) do
        data.theme[name] = pack(color)
    end
    local ok, result = pcall(function()
        return http:JSONEncode(data)
    end)
    if not ok then
        return nil, tostring(result)
    end
    return result
end

function windowmethods:import(text, silent)
    if type(text) ~= "string" then
        return false, "config must be a JSON string"
    end
    local ok, data = pcall(function()
        return http:JSONDecode(text)
    end)
    if not ok or type(data) ~= "table" or data.version ~= 1 or type(data.flags) ~= "table" then
        return false, "invalid or unsupported config"
    end
    local pending = {}
    local colors = {}
    for flag, encoded in pairs(data.flags) do
        local handle = self.controls[flag]
        if handle then
            local decoded, value = pcall(unpack, encoded)
            if not decoded or not valid(handle, value) then
                return false, "invalid value for flag '" .. tostring(flag) .. "'"
            end
            table.insert(pending, {handle = handle, value = value})
        end
    end
    if data.theme ~= nil then
        if type(data.theme) ~= "table" then
            return false, "invalid theme"
        end
        for name, encoded in pairs(data.theme) do
            if palette[name] then
                local decoded, value = pcall(unpack, encoded)
                if not decoded or typeof(value) ~= "Color3" then
                    return false, "invalid theme color '" .. tostring(name) .. "'"
                end
                colors[name] = value
            end
        end
    end
    local chosen
    if data.key ~= nil then
        local decoded, value = pcall(unpack, data.key)
        if not decoded or value ~= nil and not keycode(value) then
            return false, "invalid menu key"
        end
        chosen = value
    end
    if data.notifications ~= nil then
        local settings = data.notifications
        if type(settings) ~= "table" then
            return false, "invalid notification settings"
        end
        if settings.position ~= nil and not position(settings.position) then
            return false, "invalid notification position"
        end
        if settings.duration ~= nil and (type(settings.duration) ~= "number" or settings.duration ~= settings.duration or settings.duration < 0 or settings.duration > 3600) then
            return false, "invalid notification duration"
        end
        if settings.limit ~= nil and (type(settings.limit) ~= "number" or settings.limit ~= settings.limit or settings.limit < 1 or settings.limit > 10 or settings.limit % 1 ~= 0) then
            return false, "invalid notification limit"
        end
    end
    self:theme(colors)
    if data.notifications then
        self:configure({notifications = data.notifications})
    end
    if data.key ~= nil then
        self.key = chosen
    end
    for _, item in ipairs(pending) do
        item.handle:set(item.value, silent == true)
    end
    return true, #pending
end

function windowmethods:reset(silent)
    for _, handle in pairs(self.controls) do
        handle:set(copy(handle.default), silent == true)
    end
    return self
end

function windowmethods:save(name)
    if not safe(name) then
        return false, "config name must contain only letters, numbers, underscores or hyphens"
    end
    if type(writefile) ~= "function" then
        return false, "file storage is not available; use export instead"
    end
    local text, message = self:export()
    if not text then
        return false, message
    end
    local ok, result = pcall(function()
        if type(makefolder) == "function" then
            if type(isfolder) ~= "function" or not isfolder(self.folder) then
                pcall(makefolder, self.folder)
            end
        end
        writefile(self.folder .. "/" .. name .. ".json", text)
    end)
    if not ok then
        return false, tostring(result)
    end
    return true
end

function windowmethods:load(name, silent)
    if not safe(name) then
        return false, "config name must contain only letters, numbers, underscores or hyphens"
    end
    if type(readfile) ~= "function" then
        return false, "file storage is not available; use import instead"
    end
    local path = self.folder .. "/" .. name .. ".json"
    local ok, result = pcall(function()
        if type(isfile) == "function" and not isfile(path) then
            error("config does not exist")
        end
        return readfile(path)
    end)
    if not ok then
        return false, tostring(result)
    end
    return self:import(result, silent)
end

function windowmethods:delete(name)
    if not safe(name) then
        return false, "invalid config name"
    end
    if type(delfile) ~= "function" then
        return false, "file storage is not available"
    end
    local path = self.folder .. "/" .. name .. ".json"
    local ok, result = pcall(function()
        if type(isfile) == "function" and not isfile(path) then
            error("config does not exist")
        end
        delfile(path)
    end)
    if not ok then
        return false, tostring(result)
    end
    return true
end

function windowmethods:list()
    if type(listfiles) ~= "function" then
        return nil, "file storage is not available"
    end
    local ok, result = pcall(function()
        if type(isfolder) == "function" and not isfolder(self.folder) then
            return {}
        end
        local names = {}
        for _, path in ipairs(listfiles(self.folder)) do
            local name = path:match("([^/\\]+)%.json$")
            if safe(name) then
                table.insert(names, name)
            end
        end
        table.sort(names)
        return names
    end)
    if not ok then
        return nil, tostring(result)
    end
    return result
end

local function fit(window, list, height)
    local count = height and 1 or 0
    local total = height or 0
    for _, item in ipairs(list.items) do
        if count > 0 then
            total += 10
        end
        count += 1
        total += item.height
    end
    local available = window.root.AbsoluteSize.Y > 32 and window.root.AbsoluteSize.Y - 32 or math.huge
    while #list.items > 0 and (count > window.notifications.limit or total > available) do
        local oldest = list.items[1]
        total -= oldest.height
        if count > 1 then
            total -= 10
        end
        count -= 1
        oldest:close()
    end
end

        Size = UDim2.new(1, 0, 0, 2)
    }, card)
    paint(window, bar, "BackgroundColor3", tone)
    local handle = {closed = false, height = height}
    table.insert(list.items, handle)
    function handle:close()
        if self.closed then
            return
        end
        self.closed = true
        for index, item in ipairs(list.items) do
            if item == self then
                table.remove(list.items, index)
                break
            end
        end
        cancel(window, self.timeout)
        self.timeout = nil
        if self.timer then
            self.timer:Cancel()
        end
        if window.dead then
            return
        end
        animate(window, card, {Size = UDim2.new(1, 0, 0, 0), GroupTransparency = 1}, 0.18)
        schedule(window, window.motion and 0.2 or 0, function()
            if card.Parent then
                card:Destroy()
            end
        end)
    end
    function handle:update(value)
        if self.closed or type(value) ~= "table" then
            return self
        end
        if value.title ~= nil then
            heading.Text = tostring(value.title)
        end
        if value.text ~= nil then
            message.Text = tostring(value.text)
        end
        return self
    end
    close.Activated:Connect(function()
        handle:close()
    end)
    if action then
        local actionbutton = button(card)
        actionbutton.Position = UDim2.fromOffset(62, 86)
        actionbutton.Size = UDim2.new(1, -78, 0, 30)
        actionbutton.BackgroundTransparency = 0
        corner(actionbutton, 8)
        paint(window, actionbutton, "BackgroundColor3", "raised")
        hover(window, actionbutton, "raised", "hover")
        local actiontext = label(window, actionbutton, action.text or "Open", 12, "medium", "text")
        actiontext.Size = UDim2.fromScale(1, 1)
        actiontext.TextXAlignment = Enum.TextXAlignment.Center
        actionbutton.Activated:Connect(function()
            fire(action.callback)
            handle:close()
        end)
    end
    animate(window, card, {Size = UDim2.new(1, 0, 0, height), GroupTransparency = 0}, 0.24)
    local duration = tonumber(options.duration)
    if not duration or duration ~= duration then
        duration = window.notifications.duration
    end
    duration = math.clamp(duration, 0, 3600)
    bar.Visible = duration > 0
    if duration > 0 then
        if window.motion then
            handle.timer = tweens:Create(bar, TweenInfo.new(duration, Enum.EasingStyle.Linear), {Size = UDim2.new(0, 0, 0, 2)})
            handle.timer:Play()
        end
        handle.timeout = schedule(window, duration, function()
            handle:close()
        end)
    end
    return handle
end

function library:window(options)
    options = type(options) == "table" and options or {}
    local parent = options.parent
    if not parent then
        local player = players.LocalPlayer
        assert(player, "Glacier must run on the client")
        parent = player:WaitForChild("PlayerGui")
    end
    local window = setmetatable({
        name = tostring(options.name or "Glacier"),
        colors = copy(palette),
        paints = {},
        animations = setmetatable({}, {__mode = "k"}),
        tabs = {},
        flags = {},
        controls = {},
        binds = {},
        stacks = {},
        connections = {},
        jobs = {},
        notifications = {position = "bottom-right", duration = 4, limit = 4},
        folder = "Glacier",
        motion = options.motion ~= false,
        reopen = options.reopen ~= false,
        key = options.key == nil and Enum.KeyCode.RightShift or keycode(options.key),
        query = "",
        dead = false
    }, {__index = windowmethods})

        Name = window.name .. " Glacier",
        ResetOnSpawn = false,
        IgnoreGuiInset = true,
        DisplayOrder = tonumber(options.order) or 100,
        ZIndexBehavior = Enum.ZIndexBehavior.Sibling

        gui.ScreenInsets = Enum.ScreenInsets.DeviceSafeInsets
    end)

        Name = "Root",
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        Size = UDim2.fromScale(1, 1)

    window.root = root
    local frame = create("Frame", {
        Name = "Window",
        Active = true,
        AnchorPoint = Vector2.new(0.5, 0.5),
        BorderSizePixel = 0,
        ClipsDescendants = true,
        Position = UDim2.fromScale(0.5, 0.5),
        Size = UDim2.new(1, -32, 1, -32)
    }, root)
    window.frame = frame
    corner(frame, 17)
    paint(window, frame, "BackgroundColor3", "background")
    outline(window, frame, 0.13)

    local height = math.clamp(tonumber(options.height) or 670, 260, 1200)
    create("UISizeConstraint", {MaxSize = Vector2.new(width, height)}, frame)
    window.scale = create("UIScale", {Scale = 1}, frame)
    local rail = create("Frame", {
        Name = "Rail",
        BorderSizePixel = 0,
        Size = UDim2.new(0, 68, 1, 0)
    }, frame)
    paint(window, rail, "BackgroundColor3", "rail")

        BorderSizePixel = 0,
        BackgroundTransparency = 0.5,
        Position = UDim2.new(1, -1, 0, 0),

    }, rail)
    paint(window, railborder, "BackgroundColor3", "border")
    local brand = create("Frame", {
        AnchorPoint = Vector2.new(0.5, 0),
        BorderSizePixel = 0,
        Position = UDim2.new(0.5, 0, 0, 14),
        Size = UDim2.fromOffset(40, 40)
    }, rail)
    corner(brand, 12)
    paint(window, brand, "BackgroundColor3", "raised")
    outline(window, brand, 0.45)
    if glyph(options.logo) ~= "" then
        local symbol = picture(window, brand, options.logo, "accent")
        symbol.AnchorPoint = Vector2.new(0.5, 0.5)
        symbol.Position = UDim2.fromScale(0.5, 0.5)
        symbol.Size = UDim2.fromOffset(23, 23)
    else
        local mark = label(window, brand, window.name:sub(1, 1):upper(), 19, "bold", "accent")
        mark.Size = UDim2.fromScale(1, 1)
        mark.TextXAlignment = Enum.TextXAlignment.Center
    end
    local searchbutton = button(rail)
    searchbutton.Name = "Search"
    searchbutton.AnchorPoint = Vector2.new(0.5, 0)
    searchbutton.Position = UDim2.new(0.5, 0, 0, 68)
    searchbutton.Size = UDim2.fromOffset(44, 44)
    searchbutton.BackgroundTransparency = 1
    corner(searchbutton, 12)
    paint(window, searchbutton, "BackgroundColor3", "raised")
    local searchicon = picture(window, searchbutton, "search", "muted")
    searchicon.AnchorPoint = Vector2.new(0.5, 0.5)
    searchicon.Position = UDim2.fromScale(0.5, 0.5)
    searchicon.Size = UDim2.fromOffset(21, 21)
    searchbutton.MouseEnter:Connect(function()
        animate(window, searchbutton, {BackgroundTransparency = 0.2}, 0.14)
    end)
    searchbutton.MouseLeave:Connect(function()
        animate(window, searchbutton, {BackgroundTransparency = 1}, 0.14)
    end)
    local separator = create("Frame", {
        AnchorPoint = Vector2.new(0.5, 0),
        BackgroundTransparency = 0.5,
        BorderSizePixel = 0,
        Position = UDim2.new(0.5, 0, 0, 120),
        Size = UDim2.new(1, -28, 0, 1)
    }, rail)
    paint(window, separator, "BackgroundColor3", "border")
    local navigation = create("ScrollingFrame", {
        Name = "Tabs",
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        Position = UDim2.fromOffset(8, 132),
        Size = UDim2.new(1, -16, 1, -148),
        CanvasSize = UDim2.new(),
        AutomaticCanvasSize = Enum.AutomaticSize.Y,
        ScrollBarThickness = 0,
        ScrollingDirection = Enum.ScrollingDirection.Y,
        ElasticBehavior = Enum.ElasticBehavior.WhenScrollable
    }, rail)
    window.navigation = navigation
    create("UIListLayout", {SortOrder = Enum.SortOrder.LayoutOrder, Padding = UDim.new(0, 8)}, navigation)
    local header = create("Frame", {
        Name = "Header",
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        Position = UDim2.fromOffset(68, 0),
        Size = UDim2.new(1, -68, 0, 72)
    }, frame)
    local headerline = create("Frame", {
        BackgroundTransparency = 0.65,
        BorderSizePixel = 0,
        Position = UDim2.new(0, 0, 1, -1),
        Size = UDim2.new(1, 0, 0, 1)
    }, header)
    paint(window, headerline, "BackgroundColor3", "border")
    local drag = button(header)
    drag.Name = "Drag"
    drag.Position = UDim2.fromOffset(22, 0)
    drag.Size = UDim2.new(1, -100, 1, 0)
    local subtitle = tostring(options.subtitle or "")
    local heading = label(window, drag, window.name, 23, "bold", "text")
    heading.Size = UDim2.new(1, 0, 0, subtitle ~= "" and 32 or 72)
    heading.Position = UDim2.fromOffset(0, subtitle ~= "" and 12 or 0)
    window.heading = heading
    if subtitle ~= "" then
        local sub = label(window, drag, subtitle, 10, "medium", "muted")
        sub.Position = UDim2.fromOffset(1, 43)
        sub.Size = UDim2.new(1, 0, 0, 17)
    end
    local close = button(header)
    close.Name = "Close"
    close.AnchorPoint = Vector2.new(1, 0.5)
    close.Position = UDim2.new(1, -18, 0.5, 0)
    close.Size = UDim2.fromOffset(40, 40)
    close.BackgroundTransparency = 0
    corner(close, 11)
    outline(window, close, 0.4)
    paint(window, close, "BackgroundColor3", "raised")
    hover(window, close, "raised", "hover")
    local cross = label(window, close, "×", 23, nil, "muted")
    cross.Size = UDim2.fromScale(1, 1)
    cross.TextXAlignment = Enum.TextXAlignment.Center
    local searchholder = create("Frame", {
        Name = "Search field",
        BackgroundTransparency = 0,
        BorderSizePixel = 0,
        Position = UDim2.fromOffset(20, 16),
        Size = UDim2.new(1, -96, 0, 40),
        Visible = false,
        ZIndex = 2
    }, header)
    corner(searchholder, 10)
    paint(window, searchholder, "BackgroundColor3", "control")
    outline(window, searchholder, 0.35)
    local searchbox = create("TextBox", {
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        ClearTextOnFocus = false,
        Font = Enum.Font.Gotham,
        PlaceholderText = "Search controls...",
        Position = UDim2.fromOffset(12, 0),
        Size = UDim2.new(1, -44, 1, 0),
        Text = "",
        TextSize = 13,
        TextXAlignment = Enum.TextXAlignment.Left,
        ZIndex = 3
    }, searchholder)
    window.searchbox = searchbox
    paint(window, searchbox, "TextColor3", "text")
    paint(window, searchbox, "PlaceholderColor3", "dim")
    local searchclose = button(searchholder)
    searchclose.AnchorPoint = Vector2.new(1, 0.5)
    searchclose.Position = UDim2.new(1, -3, 0.5, 0)
    searchclose.Size = UDim2.fromOffset(34, 34)
    searchclose.ZIndex = 3
    local searchcross = label(window, searchclose, "×", 19, nil, "muted")
    searchcross.Size = UDim2.fromScale(1, 1)
    searchcross.TextXAlignment = Enum.TextXAlignment.Center
    searchcross.ZIndex = 4
    local workspace = create("Frame", {
        Name = "Workspace",
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        Position = UDim2.fromOffset(88, 90),
        Size = UDim2.new(1, -108, 1, -106)
    }, frame)
    window.workspace = workspace
    local tooltip = label(window, frame, "", 12, "medium", "text")
    tooltip.Name = "Tab tooltip"
    tooltip.AutomaticSize = Enum.AutomaticSize.X
    tooltip.BackgroundTransparency = 0
    tooltip.Size = UDim2.fromOffset(0, 30)
    tooltip.TextTruncate = Enum.TextTruncate.None
    tooltip.TextXAlignment = Enum.TextXAlignment.Center
    tooltip.Visible = false
    tooltip.ZIndex = 20
    corner(tooltip, 8)
    outline(window, tooltip, 0.35)
    paint(window, tooltip, "BackgroundColor3", "raised")
    create("UIPadding", {PaddingLeft = UDim.new(0, 10), PaddingRight = UDim.new(0, 10)}, tooltip)
    window.tooltip = tooltip
    local floating = button(root)
    floating.Name = "Reopen"
    floating.AnchorPoint = Vector2.new(0.5, 0.5)
    floating.Position = UDim2.new(1, -32, 0.5, 0)
    floating.Size = UDim2.fromOffset(50, 50)
    floating.BackgroundTransparency = 0
    floating.Visible = false
    floating.ZIndex = 10
    corner(floating, 15)
    paint(window, floating, "BackgroundColor3", "accent")
    outline(window, floating, 0.2)
    local floatingmark = label(window, floating, window.name:sub(1, 1):upper(), 18, "bold", "accenttext")
    floatingmark.Size = UDim2.fromScale(1, 1)
    floatingmark.TextXAlignment = Enum.TextXAlignment.Center
    window.floating = floating
    local function search(show)
        searchholder.Visible = show
        drag.Visible = not show
        if show then
            searchbox:CaptureFocus()
        else
            searchbox:ReleaseFocus()
            searchbox.Text = ""
            window.query = ""
            filter(window)
        end
    end
    searchbutton.Activated:Connect(function()
        search(not searchholder.Visible)
    end)
    searchclose.Activated:Connect(function()
        search(false)
    end)
    searchbox:GetPropertyChangedSignal("Text"):Connect(function()
        window.query = searchbox.Text
        filter(window)
    end)
    close.Activated:Connect(function()
        window:close()
    end)
    local moved = false
    floating.InputBegan:Connect(function(object)
        if object.UserInputType ~= Enum.UserInputType.MouseButton1 and object.UserInputType ~= Enum.UserInputType.Touch then
            return
        end
        local start = object.Position
        local origin = floating.AbsolutePosition
        moved = false
        capture(window, object, function(point)
            local dx = point.X - start.X
            local dy = point.Y - start.Y
            if math.abs(dx) + math.abs(dy) > 5 then
                moved = true
            end
            if moved then
                window.floatmoved = true
                local x = math.clamp(origin.X - root.AbsolutePosition.X + 25 + dx, 25, math.max(25, root.AbsoluteSize.X - 25))
                local y = math.clamp(origin.Y - root.AbsolutePosition.Y + 25 + dy, 25, math.max(25, root.AbsoluteSize.Y - 25))
                floating.Position = UDim2.fromOffset(x, y)
            end
        end)
    end)
    floating.Activated:Connect(function()
        if not moved then
            window:open()
        end
        moved = false
    end)
    drag.InputBegan:Connect(function(object)
        if object.UserInputType ~= Enum.UserInputType.MouseButton1 and object.UserInputType ~= Enum.UserInputType.Touch then
            return
        end
        local start = object.Position
        local origin = frame.AbsolutePosition
        capture(window, object, function(point)
            local dx = point.X - start.X
            local dy = point.Y - start.Y
            if math.abs(dx) + math.abs(dy) > 2 then
                window.moved = true
            end
            if window.moved then
                local x = math.clamp(origin.X - root.AbsolutePosition.X + dx, 0, math.max(0, root.AbsoluteSize.X - frame.AbsoluteSize.X))
                local y = math.clamp(origin.Y - root.AbsolutePosition.Y + dy, 0, math.max(0, root.AbsoluteSize.Y - frame.AbsoluteSize.Y))
                frame.Position = UDim2.fromOffset(x + frame.AbsoluteSize.X / 2, y + frame.AbsoluteSize.Y / 2)
            end
        end)
    end)
    local function bound()
        if root.AbsoluteSize.X <= 0 or root.AbsoluteSize.Y <= 0 then
            return
        end
        if window.moved and frame.AbsoluteSize.X > 0 and frame.AbsoluteSize.Y > 0 then
            local x = math.clamp(frame.AbsolutePosition.X - root.AbsolutePosition.X, 0, math.max(0, root.AbsoluteSize.X - frame.AbsoluteSize.X))
            local y = math.clamp(frame.AbsolutePosition.Y - root.AbsolutePosition.Y, 0, math.max(0, root.AbsoluteSize.Y - frame.AbsoluteSize.Y))
            frame.Position = UDim2.fromOffset(x + frame.AbsoluteSize.X / 2, y + frame.AbsoluteSize.Y / 2)
        end
        if window.floatmoved then
            local halfx = floating.AbsoluteSize.X / 2
            local halfy = floating.AbsoluteSize.Y / 2
            local x = math.clamp(floating.AbsolutePosition.X - root.AbsolutePosition.X + halfx, halfx, math.max(halfx, root.AbsoluteSize.X - halfx))
            local y = math.clamp(floating.AbsolutePosition.Y - root.AbsolutePosition.Y + halfy, halfy, math.max(halfy, root.AbsoluteSize.Y - halfy))
            floating.Position = UDim2.fromOffset(x, y)
        end
    end
    local function resize()
        local compact = root.AbsoluteSize.X < 520
        local railwidth = compact and 56 or 68
        local margin = compact and 16 or 32
        frame.Size = UDim2.new(1, -margin, 1, -margin)
        rail.Size = UDim2.new(0, railwidth, 1, 0)
        header.Position = UDim2.fromOffset(railwidth, 0)
        header.Size = UDim2.new(1, -railwidth, 0, 72)
        workspace.Position = UDim2.fromOffset(railwidth + (compact and 14 or 20), 90)
        workspace.Size = UDim2.new(1, -(railwidth + (compact and 28 or 40)), 1, -106)
        heading.TextSize = compact and 20 or 23
        bound()
        for _, list in pairs(window.stacks) do
            fit(window, list)
        end
    end
    root:GetPropertyChangedSignal("AbsoluteSize"):Connect(resize)
    frame:GetPropertyChangedSignal("AbsoluteSize"):Connect(bound)
    resize()
    table.insert(window.connections, input.InputBegan:Connect(function(object, processed)
        if window.dead or object.UserInputType ~= Enum.UserInputType.Keyboard then
            return
        end
        local code = object.KeyCode
        if window.listening then
            local binding = window.listening
            if code == Enum.KeyCode.Escape then
                binding:cancel()
            elseif code == Enum.KeyCode.Backspace or code == Enum.KeyCode.Delete then
                binding:set(nil)
            elseif code and code.Name ~= "Unknown" and code.Name ~= "None" then
                binding:set(code)
            end
            return
        end
        if processed or input:GetFocusedTextBox() then
            return
        end
        if window.key and code == window.key then
            window:visible(not frame.Visible)
            return
        end
        for _, binding in ipairs(window.binds) do
            if not binding.dead and binding.value and binding.value == code then
                if binding.mode == "hold" then
                    if not binding.held then
                        binding.held = true
                        fire(binding.callback, true, code)
                    end
                elseif binding.mode == "toggle" then
                    binding.toggled = not binding.toggled
                    fire(binding.callback, binding.toggled, code)
                else
                    fire(binding.callback, true, code)
                end
            end
        end
    end))
    table.insert(window.connections, input.InputChanged:Connect(function(object)
        local pointer = window.pointer
        if not pointer then
            return
        end
        if pointer.object.UserInputType == Enum.UserInputType.Touch and object == pointer.object or pointer.object.UserInputType == Enum.UserInputType.MouseButton1 and object.UserInputType == Enum.UserInputType.MouseMovement then
            pointer.change(object.Position)
        end
    end))
    table.insert(window.connections, input.InputEnded:Connect(function(object)
        local pointer = window.pointer
        if pointer and (pointer.object == object or pointer.object.UserInputType == Enum.UserInputType.MouseButton1 and object.UserInputType == Enum.UserInputType.MouseButton1) then
            window.pointer = nil
            if pointer.finish then
                pointer.finish()
            end
        end
        if object.UserInputType == Enum.UserInputType.Keyboard then
            for _, binding in ipairs(window.binds) do
                if binding.held and binding.value == object.KeyCode then
                    binding.held = false
                    fire(binding.callback, false, binding.value)
                end
            end
        end
    end))
    window:configure(options)
    table.insert(windows, window)
    return window
end

function library:destroy()
    while #windows > 0 do
        windows[#windows]:destroy()
    end
end

return library
