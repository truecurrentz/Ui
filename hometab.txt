local Env = getgenv()
local Players = game:GetService("Players")
local Player = Players.LocalPlayer

local Lumen = Env.Lumen
if not Lumen then
	Lumen = loadstring(game:HttpGet("https://raw.githubusercontent.com/truecurrentz/XxUixX/main/Lumen.lua.txt"))()
end

local HomeCfg = (typeof(Env.LumenHome) == "table" and Env.LumenHome) or {}
local DiscordInvite = (typeof(HomeCfg.DiscordInvite) == "string" and HomeCfg.DiscordInvite) or "https://discord.gg/CwsJz3uyEZ"

local SupportedGames = {
	{ Name = "Game one"; Status = "Working" };
	{ Name = "Game two"; Status = "Testing" };
}

local Updates = (typeof(HomeCfg.Updates) == "table" and HomeCfg.Updates) or {
	{ Title = "v1.0"; Body = "First release." };
}

local IsKeyless = (typeof(JD_REASON) == "string" and JD_REASON == "KEYLESS") or false
local IsPremium = (typeof(JD_IS_PREMIUM) == "boolean" and JD_IS_PREMIUM) or false
local ExpiresAt = (typeof(JD_EXPIRES_AT) == "number" and JD_EXPIRES_AT) or nil

if HomeCfg.IsKeyless == true then
	IsKeyless = true
end
if HomeCfg.IsPremium == true then
	IsPremium = true
end
if IsKeyless == true then
	IsPremium = false
end
if type(ExpiresAt) ~= "number" and type(HomeCfg.ExpiresAt) == "number" then
	ExpiresAt = HomeCfg.ExpiresAt
end

if Lumen.JunkieResult then
	if Lumen.JunkieResult.is_keyless == true then
		IsKeyless = true
	end
	if Lumen.JunkieResult.is_premium == true and not IsKeyless then
		IsPremium = true
	end
end

if type(ExpiresAt) ~= "number" and Lumen.Auth and type(Lumen.Auth.ExpiresAt) == "number" then
	ExpiresAt = Lumen.Auth.ExpiresAt
end

local KeyText = "Not linked"
local RawKey = nil
if typeof(Env.LumenUserKey) == "string" and Env.LumenUserKey ~= "" then
	RawKey = Env.LumenUserKey
elseif typeof(Env.SCRIPT_KEY) == "string" and Env.SCRIPT_KEY ~= "" then
	RawKey = Env.SCRIPT_KEY
end
if RawKey then
	KeyText = RawKey
elseif IsKeyless then
	KeyText = "KEYLESS"
end

local Headshot = ""
pcall(function()
	Headshot = Players:GetUserThumbnailAsync(Player.UserId, Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size150x150)
end)
if type(Headshot) ~= "string" or Headshot == "" then
	Headshot = "rbxthumb://type=AvatarHeadShot&id=" .. tostring(Player.UserId) .. "&w=150&h=150"
end

local function ExpiryText()
	if IsKeyless then
		return "Keyless session"
	end
	if type(ExpiresAt) == "number" then
		if ExpiresAt - os.time() <= 0 then
			return "Expired"
		end
		return Lumen.FormatExpiry(ExpiresAt) .. " left"
	end
	return "Lifetime"
end

local function LiveInfo()
	local Names = {}
	local Count = 0
	if typeof(JD_LIVE_USERS) == "table" then
		for _, Id in ipairs(JD_LIVE_USERS) do
			local Plr = Players:GetPlayerByUserId(Id)
			if Plr then
				Count += 1
				table.insert(Names, Plr.DisplayName)
			end
		end
	end
	if Count == 0 then
		return 1, { Player.DisplayName }
	end
	return Count, Names
end

local function StatusText()
	if IsPremium then
		return "Premium"
	end
	if IsKeyless then
		return "Keyless"
	end
	return "Standard"
end

local function BuildHome(Window)
	Window = Window or Env.LumenHomeWindow
	if not Window then
		Window = Lumen:Window({
			Title = "Lumen";
			Footer = ".gg/yourserver";
			Logo = "layers";
		})
		Env.LumenHomeWindow = Window
	end

	if type(ExpiresAt) == "number" and Lumen.SetKeyTime then
		pcall(Lumen.SetKeyTime, ExpiresAt)
	end

	local HomePage = Window:Page({ Icon = "home" })
	local HomeTab = HomePage:SubPage({ Name = "Home" })

	local Profile = HomeTab:Section({ Name = "Profile"; Side = "Left"; Icon = "user" })

	Profile:Image({ Image = Headshot; Height = 120 })

	Profile:Paragraph({
		Title = Player.DisplayName;
		Body = "@" .. Player.Name .. "\nStatus: " .. StatusText();
	})

	Profile:Paragraph({
		Title = "Key";
		Body = KeyText;
	})

	local TimePara = Profile:Paragraph({
		Title = "Key time";
		Body = ExpiryText();
	})

	local GamesSection = HomeTab:Section({ Name = "Supported games"; Side = "Right"; Icon = "gamepad" })

	for _, GameInfo in ipairs(SupportedGames) do
		local GameName = GameInfo
		local GameStatus = ""
		if type(GameInfo) == "table" then
			GameName = tostring(GameInfo.Name or "Game")
			GameStatus = tostring(GameInfo.Status or "")
		end
		if GameStatus == "" then
			GameStatus = "Supported"
		end
		GamesSection:Paragraph({
			Title = GameName;
			Body = GameStatus;
		})
	end

	local UpdatesSection = HomeTab:Section({ Name = "Latest updates"; Side = "Left"; Icon = "bell" })

	for _, UpdateInfo in ipairs(Updates) do
		local UpdateTitle = "Update"
		local UpdateBody = tostring(UpdateInfo)
		if type(UpdateInfo) == "table" then
			UpdateTitle = tostring(UpdateInfo.Title or "Update")
			UpdateBody = tostring(UpdateInfo.Body or "")
		end
		UpdatesSection:Paragraph({
			Title = UpdateTitle;
			Body = UpdateBody;
		})
	end

	local LiveSection = HomeTab:Section({ Name = "Live users"; Side = "Right"; Icon = "users" })

	local LiveCount, LiveNames = LiveInfo()
	local LivePara = LiveSection:Paragraph({
		Title = "On right now";
		Body = tostring(LiveCount) .. " live: " .. table.concat(LiveNames, ", ");
	})

	LiveSection:Button({
		Name = "Join discord";
		Callback = function()
			local Opened = false
			pcall(function()
				local web = cloneref(game:GetService("WebViewService"))
				web:OpenWindowV2(DiscordInvite)
				Opened = true
				task.delay(9, function()
					pcall(function()
						web:CloseWindow()
					end)
				end)
			end)
			if not Opened and typeof(setclipboard) == "function" then
				Opened = pcall(setclipboard, DiscordInvite) == true
			end
			Lumen.Notify({
				Title = "Discord";
				Text = Opened and "Opening the discord invite." or "Could not open the invite on this executor.";
				Type = Opened and "Success" or "Error";
				Duration = 3;
			})
		end;
	})

	task.spawn(function()
		while task.wait(30) do
			local Alive = LivePara and LivePara.Frame and LivePara.Frame.Parent
			if not Alive then
				break
			end
			local NewCount, NewNames = LiveInfo()
			LivePara.SetBody(tostring(NewCount) .. " live: " .. table.concat(NewNames, ", "))
			if TimePara and TimePara.SetBody then
				TimePara.SetBody(ExpiryText())
			end
		end
	end)

	return HomeTab
end

if not Env.LumenHomeBuilt then
	Env.LumenHomeBuilt = true
	BuildHome(Env.LumenHomeWindow)
end

return BuildHome
