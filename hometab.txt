
local Env = getgenv()
local Plrs = game:GetService("Players")
local Http = game:GetService("HttpService")
local Plr = Plrs.LocalPlayer

local Lum = Env.Lumen
if not Lum then
	Lum = loadstring(game:HttpGet("https://raw.githubusercontent.com/truecurrentz/XxUixX/main/Lumen.lua.txt"))()
end

local Cfg = (typeof(Env.LumenHome) == "table" and Env.LumenHome) or {}
local Invite = (typeof(Cfg.DiscordInvite) == "string" and Cfg.DiscordInvite) or "https://discord.gg/CwsJz3uyEZ"

local Games = (typeof(Cfg.SupportedGames) == "table" and Cfg.SupportedGames) or {
	{ Name = "Saber Showdown"; Status = "Working" };
	{ Name = "Bug fixes"; Status = "Working" };
}

local Updates = (typeof(Cfg.Updates) == "table" and Cfg.Updates) or {
	{ Title = "[+] New ui"; Body = "New UI update." };
}

local Keyless = (typeof(JD_REASON) == "string" and JD_REASON == "KEYLESS") or false
local Premium = (typeof(JD_IS_PREMIUM) == "boolean" and JD_IS_PREMIUM) or false
local Exp = (typeof(JD_EXPIRES_AT) == "number" and JD_EXPIRES_AT) or nil

if Cfg.IsKeyless == true then
	Keyless = true
end

if Cfg.IsPremium == true then
	Premium = true
end

if Keyless then
	Premium = false
end

if type(Exp) ~= "number" and type(Cfg.ExpiresAt) == "number" then
	Exp = Cfg.ExpiresAt
end

if Lum.JunkieResult then
	if Lum.JunkieResult.is_keyless == true then
		Keyless = true
	end
	if Lum.JunkieResult.is_premium == true and not Keyless then
		Premium = true
	end
end

if type(Exp) ~= "number" and Lum.Auth and type(Lum.Auth.ExpiresAt) == "number" then
	Exp = Lum.Auth.ExpiresAt
end

local Key = "Not linked"
local Raw = nil

if typeof(Env.LumenUserKey) == "string" and Env.LumenUserKey ~= "" then
	Raw = Env.LumenUserKey
elseif typeof(Env.SCRIPT_KEY) == "string" and Env.SCRIPT_KEY ~= "" then
	Raw = Env.SCRIPT_KEY
end

if Raw then
	Key = Raw
elseif Keyless then
	Key = "KEYLESS"
end

local Head = ""

pcall(function()
	Head = Plrs:GetUserThumbnailAsync(
		Plr.UserId,
		Enum.ThumbnailType.HeadShot,
		Enum.ThumbnailSize.Size150x150
	)
end)

if type(Head) ~= "string" or Head == "" then
	Head = "rbxthumb://type=AvatarHeadShot&id=" .. tostring(Plr.UserId) .. "&w=150&h=150"
end

local function Code()
	if typeof(Invite) ~= "string" then
		return nil
	end

	local Val = Invite:match("discord%.gg/([%w%-]+)")
	if not Val then
		Val = Invite:match("discord%.com/invite/([%w%-]+)")
	end

	return Val
end

local function DCount()
	if typeof(Cfg.DiscordUsers) == "number" then
		return Cfg.DiscordUsers
	end

	local Val = Code()
	if not Val then
		return nil
	end

	local Count = nil

	pcall(function()
		local Res = game:HttpGet("https://discord.com/api/v9/invites/" .. Val .. "?with_counts=true")
		local Data = Http:JSONDecode(Res)

		if type(Data) == "table" and type(Data.approximate_member_count) == "number" then
			Count = Data.approximate_member_count
		end
	end)

	return Count
end

local function Status()
	if Premium then
		return "Premium"
	end
	if Keyless then
		return "Keyless"
	end
	return "Standard"
end

local function Alive(Obj)
	return typeof(Obj) == "Instance" and Obj.Parent ~= nil
end

local function WinAlive(Win)
	return type(Win) == "table" and Alive(Win.Canvas)
end

local function HomeAlive(Win)
	if not WinAlive(Win) then
		return false
	end

	local Page = Win.LumenHomePage
	return type(Page) == "table" and Alive(Page.Button)
end

local function Pick(Win)
	if WinAlive(Win) then
		return Win
	end

	if WinAlive(Env.LumenHomeWindow) then
		return Env.LumenHomeWindow
	end

	if type(Lum.Windows) == "table" then
		for _, Val in ipairs(Lum.Windows) do
			if WinAlive(Val) then
				return Val
			end
		end
	end

	return nil
end

local function Build(Win)
	Win = Pick(Win)

	if not Win then
		Win = Lum:Window({
			Title = "Lumen";
			Footer = ".gg/yourserver";
			Logo = "layers";
		})
	end

	Env.LumenHomeWindow = Win

	if HomeAlive(Win) then
		return Win.LumenHomeTab
	end

	if type(Exp) == "number" and Lum.SetKeyTime then
		pcall(Lum.SetKeyTime, Exp)
	end

	local Page = Win:Page({
		Icon = "home";
	})

	local Tab = Page:SubPage({
		Name = "Home";
	})

	Win.LumenHomePage = Page
	Win.LumenHomeTab = Tab

	local Prof = Tab:Section({
		Name = "Profile";
		Side = "Left";
		Icon = "user";
	})

	Prof:Image({
		Image = Head;
		Height = 120;
	})

	Prof:Paragraph({
		Title = Plr.DisplayName;
		Body = "@" .. Plr.Name .. "\nStatus: " .. Status();
	})

	Prof:Paragraph({
		Title = "Key";
		Body = Key;
	})

	local GSec = Tab:Section({
		Name = "Supported games";
		Side = "Right";
		Icon = "gamepad";
	})

	for _, Val in ipairs(Games) do
		local Name = Val
		local State = ""

		if type(Val) == "table" then
			Name = tostring(Val.Name or "Game")
			State = tostring(Val.Status or "")
		end

		if State == "" then
			State = "Supported"
		end

		GSec:Paragraph({
			Title = Name;
			Body = State;
		})
	end

	local Usec = Tab:Section({
		Name = "Latest updates";
		Side = "Left";
		Icon = "bell";
	})

	for _, Val in ipairs(Updates) do
		local Name = "Update"
		local Body = tostring(Val)

		if type(Val) == "table" then
			Name = tostring(Val.Title or "Update")
			Body = tostring(Val.Body or "")
		end

		Usec:Paragraph({
			Title = Name;
			Body = Body;
		})
	end

	local DSec = Tab:Section({
		Name = "Discord users";
		Side = "Right";
		Icon = "users";
	})

	DSec:Button({
		Name = "Join discord";
		Callback = function()
			local Open = false

			pcall(function()
				local Web = cloneref(game:GetService("WebViewService"))
				Web:OpenWindowV2(Invite)
				Open = true

				task.delay(9, function()
					pcall(function()
						Web:CloseWindow()
					end)
				end)
			end)

			if not Open and typeof(setclipboard) == "function" then
				Open = pcall(setclipboard, Invite) == true
			end

			Lum.Notify({
				Title = "Discord";
				Text = Open and "Opening the discord invite." or "Could not open the invite on this executor.";
				Type = Open and "Success" or "Error";
				Duration = 3;
			})
		end;
	})

	local Count = DCount()

	local DPara = DSec:Paragraph({
		Title = "Discord members";
		Body = Count and tostring(Count) or "Unavailable";
	})

	task.spawn(function()
		while task.wait(120) do
			if not WinAlive(Win) or not DPara or not Alive(DPara.Frame) then
				break
			end

			local New = DCount()
			DPara.SetBody(New and tostring(New) or "Unavailable")
		end
	end)

	return Tab
end

Env.LumenBuildHome = Build

Build(Env.LumenHomeWindow)

return Build
